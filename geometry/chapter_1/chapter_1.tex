	\mathblock{definition}{Variété topologique}{
		Soit $M$ un espace topologique. \\
		Soit $n \in \mathbb{N}^{\star}$. \\

		On dit que $M$ est une \textit{variété topologique de dimension $n$} lorsque

		\begin{mathlist}
			\item $M$ est séparable (ou \textsc{Hausdorff}).
			\item $M$ admet un base dénombrable.
			\item $M$ est localement euclidien à $n$ dimensions :
				$$
					\forall p \in M, \exists U \in \mathcal{O}_{p}, \exists V \in \mathcal{O}_{\mathbb{R}^{n}}, \exists \varphi : U \longrightarrow V \ \text{homéomorphe}
				$$
		\end{mathlist}
	}
